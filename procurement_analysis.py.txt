import datetime
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd
import seaborn as sns

# Set visual styling
sns.set_theme(style="whitegrid")
np.random.seed(42)

# --- 1. GENERATE SAMPLE SA MUNICIPAL PROCUREMENT DATA ---
suppliers = [
    "Khanya Logistics CC",
    "Maluti Industrial Supplies",
    "Amandla Tech Solutions",
    "Platberg Spares",
    "Vanguard Civil Engineering",
]
categories = [
    "Fleet Maintenance",
    "IT Equipment",
    "Office Consumables",
    "Civil Infrastructure",
    "Protective Gear",
]
municipal_departments = [
    "Infrastructure Services",
    "Community Services",
    "Corporate Governance",
    "Finance & Admin",
]

n_records = 200

data = {
    "PO_Number": [f"PO-2026-{i:04d}" for i in range(1, n_records + 1)],
    "Department": np.random.choice(municipal_departments, n_records),
    "Supplier_Name": np.random.choice(
        suppliers, n_records, p=[0.35, 0.25, 0.15, 0.15, 0.10]
    ),
    "Category": np.random.choice(categories, n_records),
    "Order_Date": pd.date_range(
        start="2026-01-01", periods=n_records, freq="D"
    ),
    "Amount_ZAR": np.random.choice(
        [185000, 195000, 198000, 450000, 12000, 85000, 620000],
        n_records,
        p=[0.20, 0.25, 0.15, 0.10, 0.15, 0.10, 0.05],
    ),
}

df = pd.DataFrame(data)

# --- 2. DATA AUDIT: COMPLIANCE & RISK CHECKS ---
df["MFMA_Audit_Flag"] = df["Amount_ZAR"].apply(
    lambda x: (
        "High Risk (Near R200k Limit)"
        if 180000 <= x < 200000
        else "Standard Order"
    )
)

print("--- AUDIT SUMMARY: TRANSACTION RISK FLAGS ---")
print(df["MFMA_Audit_Flag"].value_counts())
print("\n" + "=" * 50 + "\n")

supplier_summary = (
    df.groupby("Supplier_Name")
    .agg(
        Total_Spend_ZAR=("Amount_ZAR", "sum"),
        Order_Count=("PO_Number", "count"),
        Flagged_Orders=(
            "MFMA_Audit_Flag",
            lambda x: (x == "High Risk (Near R200k Limit)").sum(),
        ),
    )
    .reset_index()
    .sort_values(by="Total_Spend_ZAR", ascending=False)
)

print("--- SUPPLIER SPEND & AUDIT SUMMARY ---")
print(supplier_summary.to_string(index=False))

# --- 3. DATA VISUALIZATION ---
fig, axes = plt.subplots(1, 2, figsize=(14, 5))

sns.barplot(
    data=supplier_summary,
    x="Total_Spend_ZAR",
    y="Supplier_Name",
    palette="Blues_r",
    ax=axes[0],
)
axes[0].set_title(
    "Total Procurement Spend by Supplier (ZAR)", fontsize=12, fontweight="bold"
)
axes[0].set_xlabel("Total Spend (ZAR)")
axes[0].set_ylabel("Supplier")

sns.histplot(
    data=df, x="Amount_ZAR", bins=15, kde=True, color="darkblue", ax=axes[1]
)
axes[1].axvline(
    200000, color="red", linestyle="--", label="R200,000 Tender Threshold"
)
axes[1].set_title(
    "Purchase Order Value Distribution", fontsize=12, fontweight="bold"
)
axes[1].set_xlabel("PO Amount (ZAR)")
axes[1].legend()

plt.tight_layout()
plt.savefig("procurement_audit_summary.png")
plt.show()
